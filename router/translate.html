Drop-in i18n Snippet

      

        Add this script tag to any HTML page. It will auto-translate all elements with data-i18n attributes using the muLLM translation API.
      

      

        
<!-- Add to your HTML <head> -->
<script src="https://10.0.0.165:6856/api/i18n.js?lang=es"></script>

<!-- Then mark elements: -->
<h1 data-i18n="greeting">Hello</h1>
<p data-i18n="description">This text will be translated.</p>
      


      
Programmatic API

      

        
// POST /api/translate
const res = await fetch('https://10.0.0.165:6856/api/translate', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    text: 'Hello world',
    target_lang: 'es',
    source_lang: 'auto'  // optional
  })
});
const { translation, source_lang, cost } = await res.json();
      


      
Batch API

      

        
// POST /api/translate/batch — translate many strings at once
const res = await fetch('https://10.0.0.165:6856/api/translate/batch', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    strings: { greeting: 'Hello', farewell: 'Goodbye' },
    target_lang: 'fr'
  })
});
const { translations } = await res.json(); // { greeting: 'Bonjour', farewell: 'Au revoir' }
